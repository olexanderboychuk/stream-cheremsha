## The Application (Server) Code

import sys
import threading
import socket
from PyQt6.QtWidgets import QApplication, QWidget, QVBoxLayout, QPushButton, QLineEdit, QLabel
from PyQt6.QtCore import pyqtSignal, QObject
from PyQt6.QtTest import QTest
from PyQt6.QtCore import Qt

# 1. Create a Worker to handle the Server Logic separately
class CommandServer(QObject):
    command_received = pyqtSignal(str) # Signal to send command to Main Thread

    def start_server(self):
        # Run socket server in a standard Python thread so it doesn't freeze GUI
        t = threading.Thread(target=self._server_loop, daemon=True)
        t.start()

    def _server_loop(self):
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server.bind(('localhost', 5000))
        server.listen(1)
        print("Server listening on port 5000...")
        
        while True:
            client, addr = server.accept()
            data = client.recv(1024).decode('utf-8')
            if data:
                # Emit signal to Main Thread to perform the UI action
                self.command_received.emit(data)
                client.send(b"OK")
            client.close()

# 2. The Main Application
class TargetApp(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Test Target App")
        self.resize(300, 200)
        
        # UI Elements
        self.layout = QVBoxLayout()
        
        self.status_label = QLabel("Waiting...")
        self.status_label.setObjectName("status_lbl") # ID for finding it
        
        self.input_field = QLineEdit()
        self.input_field.setObjectName("my_input")    # ID for finding it
        
        self.btn = QPushButton("Submit")
        self.btn.setObjectName("my_btn")              # ID for finding it
        self.btn.clicked.connect(lambda: self.status_label.setText(f"Submitted: {self.input_field.text()}"))

        self.layout.addWidget(self.status_label)
        self.layout.addWidget(self.input_field)
        self.layout.addWidget(self.btn)
        self.setLayout(self.layout)

        # Start Server
        self.server = CommandServer()
        self.server.command_received.connect(self.handle_command)
        self.server.start_server()

    # This runs on the Main UI Thread
    def handle_command(self, cmd_str):
        print(f"Executing: {cmd_str}")
        parts = cmd_str.split(':')
        action = parts[0]

        if action == "click":
            # Find widget by objectName and click it
            widget_name = parts[1]
            target = self.findChild(QWidget, widget_name)
            if target:
                QTest.mouseClick(target, Qt.MouseButton.LeftButton)

        elif action == "type":
            # Find widget and type text
            widget_name = parts[1]
            text_to_type = parts[2]
            target = self.findChild(QWidget, widget_name)
            if target:
                # Clear first (optional)
                target.clear()
                QTest.keyClicks(target, text_to_type)
        
        elif action == "screenshot":
            # Save screenshot of the app
            filename = parts[1]
            self.grab().save(filename)
            print(f"Screenshot saved to {filename}")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = TargetApp()
    window.show()
    sys.exit(app.exec())

import socket
import time

def send_command(cmd):
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.connect(('localhost', 5000))
    s.send(cmd.encode('utf-8'))
    print(f"Sent: {cmd} -> Response: {s.recv(1024).decode()}")
    s.close()
    time.sleep(0.5) # Wait for animation/update

# 1. Type into the box
send_command("type:my_input:Hello World")

# 2. Click the button
send_command("click:my_btn")

# 3. Take a screenshot to verify

import sys
from PyQt6.QtWidgets import QApplication, QPushButton, QVBoxLayout, QWidget, QLabel
from PyQt6.QtGui import QPixmap

class ScreenshotApp(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Self Screenshot App")
        self.resize(300, 200)

        layout = QVBoxLayout()
        self.label = QLabel("Click the button to save me!")
        self.btn = QPushButton("Take Screenshot")
        self.btn.clicked.connect(self.take_screenshot)

        layout.addWidget(self.label)
        layout.addWidget(self.btn)
        self.setLayout(layout)

    def take_screenshot(self):
        # 1. Grab the specific widget (self = the whole window)
        # This renders the widget directly to a pixmap
        screenshot = self.grab() 
        
        # 2. Save to file
        screenshot.save("myscreenshot.png")
        self.label.setText("Saved to myscreenshot.png")
        print("Screenshot saved.")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = ScreenshotApp()
    window.show()
    sys.exit(app.exec())
